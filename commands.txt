# ─── MergeReturn Pass — Manual Commands ────────────────────────────────────
#
# All commands assume you are in the repository root and that
# clang++, opt, and llvm-config point to the same LLVM installation.
#
# Tested with LLVM 14 / 16 (legacy pass manager).
# ────────────────────────────────────────────────────────────────────────────

# 1. Compile the pass into a shared library
clang++ -fPIC -g -O2 -fno-rtti -shared \
    $(llvm-config --cppflags --ldflags --libs) \
    -o MergeReturnPass.so \
    MergeReturnPass/MergeReturnPass.cpp

# 2. Compile the test program to human-readable LLVM IR
clang++ -S -fno-discard-value-names -emit-llvm -O0 \
        -Xclang -disable-O0-optnone \
        Tests/test.cpp -o before.ll

# 3. Apply the mergereturn pass
opt -S -enable-new-pm=0 \
    -load ./MergeReturnPass.so \
    -mergereturn \
    before.ll -o after.ll

# 4. (Optional) View the CFG of a function before and after
opt -S -enable-new-pm=0 -dot-cfg before.ll -o /dev/null
opt -S -enable-new-pm=0 -dot-cfg after.ll  -o /dev/null
dot -Tpng .max.dot -o max_before.png
# re-run after the pass for max_after.png

# 5. Compile the transformed IR to a native binary and run it
clang++ after.ll -o test_optimised
./test_optimised

# 6. Verify: count ret instructions
grep -c '^\s*ret ' before.ll     # e.g. 9  (multiple per function)
grep -c '^\s*ret ' after.ll      # e.g. 5  (one per function)

# ── CMake build (alternative) ────────────────────────────────────────────────
mkdir -p build && cd build
cmake .. -DLLVM_DIR=$(llvm-config --cmakedir)
make
# produces MergeReturnPass.so in ./build/
